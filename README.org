* INTRO

  A minimal PostgreSQL C extension (=whatever=) implementing a custom =complex=
  number data type (=complex_in= / =complex_out=), based on the workshop
  [[https://2026.pgconf.org.np/speaker/users/38275741-9346-454c-8a07-75cf8b1dfd28][Write PostgreSQL Extensions in C]] at [[https://2026.pgconf.org.np][pgconf.org.np]].

  Prerequisites
  - PostgreSQL server
  - development headers (=pg_config= / =PGXS=),
  - =gcc=
  - =make=

* INSTALL

  Installing into PostgreSQL's system directories requires elevated
  privileges

  #+HEADER: :results output :eval no-export
  #+BEGIN_SRC sh :exports both :dir (concat "/sudo::" (expand-file-name "."))
    2>&1 make install
  #+END_SRC

  #+RESULTS:
  #+begin_example
  /usr/bin/mkdir -p '/usr/lib/postgresql'
  /usr/bin/mkdir -p '/usr/share/postgresql/extension'
  /usr/bin/mkdir -p '/usr/share/postgresql/extension'
  /usr/bin/install -c -m 755  whatever.so '/usr/lib/postgresql/whatever.so'
  /usr/bin/install -c -m 644 .//whatever.control '/usr/share/postgresql/extension/'
  /usr/bin/install -c -m 644 .//whatever--1.0.sql  '/usr/share/postgresql/extension/'
  /usr/bin/mkdir -p '/usr/lib/postgresql/bitcode/whatever'
  /usr/bin/mkdir -p '/usr/lib/postgresql/bitcode'/whatever/
  /usr/bin/install -c -m 644 whatever.bc '/usr/lib/postgresql/bitcode'/whatever/./
  cd '/usr/lib/postgresql/bitcode' && /usr/bin/llvm-lto -thinlto -thinlto-action=thinlink -o whatever.index.bc whatever/whatever.bc
  #+end_example

  NOTE: after rebuilding and reinstalling =whatever.so=, a new
  PostgreSQL session is required to load the updated shared library

  #+HEADER: :cmdline whatever
  #+HEADER: :prologue "\\pset linestyle ascii\\pset format aligned"
  #+HEADER: :results output :eval no-export
  #+BEGIN_SRC sql :engine postgresql :exports both
    DROP EXTENSION whatever;
    CREATE EXTENSION IF NOT EXISTS whatever;
  #+END_SRC

  #+RESULTS:
  : DROP EXTENSION
  : CREATE EXTENSION

* EXTENSION
** complex-in

   #+HEADER: :cmdline whatever
   #+HEADER: :results output table :eval no-export
   #+BEGIN_SRC sql :engine postgresql :exports both
     SELECT complex_in('(3, 4)');
   #+END_SRC

   #+RESULTS:
   | complex_in |
   |------------|
   | (3,4)      |


** complex-out

   #+HEADER: :cmdline whatever
   #+HEADER: :results output table :eval no-export
   #+BEGIN_SRC sql :engine postgresql :exports both
     SELECT complex_out('(3, 4)'::complex);
   #+END_SRC

   #+RESULTS:
   | complex_out |
   |-------------|
   | (3,4)       |

** type usage

   #+HEADER: :cmdline whatever
   #+HEADER: :results output table :eval no-export
   #+BEGIN_SRC sql :engine postgresql :exports both
     SELECT '(3, 4)'::complex AS value, pg_typeof('(3, 4)'::complex) AS type;
   #+END_SRC

   #+RESULTS:
   | value | type    |
   |-------+---------|
   | (3,4) | complex |
